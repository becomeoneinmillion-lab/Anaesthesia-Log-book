Google Drive Cross-Device Sync Setup

Follow these 5 simple steps to connect your personal Google Drive. Once connected, your logbook will sync automatically across your phone, tablet, and computer using your single Web App URL.

1
Open Google Apps Script

Visit Google Apps Script in your browser using the button below:

Open script.google.com

Click on "+ New Project" in the top-left corner. Name the project Anaesthesiologist Logbook Sync.

2
Paste the Backend Code

Delete everything in the script editor, and paste the code from the google_apps_script_code.txt file into Code.gs.

Code.gs Preview 
const FOLDER_NAME = "Anaesthesiologist Logbook";
const DB_FILE_NAME = "logbook_database.json";
function getOrCreateFolder() {
const folders = DriveApp.getFoldersByName(FOLDER_NAME);
if (folders.hasNext()) return folders.next();
return DriveApp.createFolder(FOLDER_NAME);
}
function getDatabaseFile() {
const folder = getOrCreateFolder();
const files = folder.getFilesByName(DB_FILE_NAME);
if (files.hasNext()) return files.next();
return folder.createFile(DB_FILE_NAME, JSON.stringify({ cases: [], doctorProfile: {} }), "application/json");
}
function doGet(e) {
try {
const dbFile = getDatabaseFile();
const content = dbFile.getBlob().getDataAsString();
return ContentService.createTextOutput(content).setMimeType(ContentService.MimeType.JSON);
} catch (err) {
return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
}
}
function doPost(e) {
try {
if (!e || !e.postData || !e.postData.contents) return responseJson({ status: "error", message: "No data" });
const payload = JSON.parse(e.postData.contents);
const folder = getOrCreateFolder();
if (payload.testConnection) {
return responseJson({ status: "success", message: "Drive access confirmed", url: folder.getUrl() });
}
if (payload.action === "syncDatabase" || payload.allCases) {
const dbFile = getDatabaseFile();
const dbData = { updatedAt: new Date().toISOString(), doctorProfile: payload.doctorProfile || {}, cases: payload.allCases || payload.cases || [] };
dbFile.setContent(JSON.stringify(dbData, null, 2));
return responseJson({ status: "success", message: "Database updated on Google Drive", totalCases: dbData.cases.length });
}
if (payload.action === "saveCsv" || payload.csvContent) {
const fName = payload.fileName || ("Anaesthesiologist_Logbook_" + new Date().toISOString().split("T")[0] + ".csv");
const csvFile = folder.createFile(fName, payload.csvContent, "text/csv;charset=utf-8");
csvFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
return responseJson({ status: "success", viewUrl: csvFile.getUrl(), fileName: fName });
}
const b64 = payload.base64Data || payload.base64 || payload.data;
if (b64) {
const fName = payload.fileName || ("File_" + Date.now() + ".jpg");
const mime = payload.mimeType || "image/jpeg";
const blob = Utilities.newBlob(Utilities.base64Decode(b64), mime, fName);
const file = folder.createFile(blob);
file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
return responseJson({ status: "success", viewUrl: file.getUrl(), fileId: file.getId() });
}
return responseJson({ status: "error", message: "Unknown action" });
} catch (err) {
return responseJson({ status: "error", error: err.toString() });
}
}
function responseJson(obj) {
return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
3
Deploy as a Web App

1. In the Apps Script toolbar, click Deploy → New deployment.

2. Click the gear icon () and choose Web app.

3. Set the deployment fields accurately:

Mandatory Settings: • Execute as: Me • Who has access: Anyone (Selecting "Anyone" allows your phones and tablets to sync seamlessly).

4. Click Deploy.

4
Grant Google Permission

1. Click Authorize access and select your Google account.

2. If you see "Google hasn't verified this app", click Advanced → Go to Anaesthesiologist Logbook Sync (unsafe).

3. Click Allow.

5
Paste Web App URL into Settings

Copy the Web app URL (ends in /exec) and paste it into the Settings tab.

On any other device, paste that same URL to load all your records instantly!
